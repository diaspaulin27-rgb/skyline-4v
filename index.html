CONFIGURAÇÕES GERAIS E ESTADO DO JOGO (GAME STATE) ------------------------------------------------------------------------------- const CONFIG = { TOTAL_LEVELS: 20, WIN_THRESHOLD: 0.99, // Alterado de 0.72 para 0.99 (99% de limpeza necessária) PERFECT_CLEAR_THRESHOLD: 1.0, // Requer 100% de área limpa PERFECT_CLEAR_TIME_LIMIT: 8.0, // Limite de tempo em segundos para acionar "Limpeza Perfeita" BRUSH_RADIUS: 25, SNOW_LEVELS: [11, 12, 13, 14, 15] // Fases de neve com mecânica em 2 etapas }; let gameState = { currentLevel: parseInt(localStorage.getItem('lastSelectedLevel')) || 1, unlockedLevels: parseInt(localStorage.getItem('unlockedLevels')) || 1, isPaused: false, clearedPercentage: 0.0, startTime: 0, isLevelComplete: false }; ------------------------------------------------------------------------------- 2. INICIALIZAÇÃO DO MENU PRINCIPAL & SELEÇÃO DE FASES ------------------------------------------------------------------------------- function initMainMenu() { // Carrega a fase atual ou a última que o jogador selecionou (fim da seleção aleatória) let savedLevel = parseInt(localStorage.getItem('lastSelectedLevel')); if (!savedLevel || savedLevel < 1 || savedLevel > CONFIG.TOTAL_LEVELS) { savedLevel = gameState.unlockedLevels; } gameState.currentLevel = savedLevel; renderLevelSelectorUI(savedLevel); } function renderLevelSelectorUI(activeLevel) { console.log(`Exibindo no Menu Principal: Fase ${activeLevel} de ${CONFIG.TOTAL_LEVELS}`); // Atualiza a interface gráfica do Menu para destacar a fase selecionada const levelDisplayElement = document.getElementById('selected-level-card'); if (levelDisplayElement) { levelDisplayElement.innerText = `Fase ${activeLevel}`; } } function selectLevel(levelNumber) { if (levelNumber >= 1 && levelNumber <= gameState.unlockedLevels) { gameState.currentLevel = levelNumber; localStorage.setItem('lastSelectedLevel', levelNumber); renderLevelSelectorUI(levelNumber); } } ------------------------------------------------------------------------------- 3. CARREGAMENTO E FLUXO DAS 20 FASES ------------------------------------------------------------------------------- function startLevel(levelNumber) { if (levelNumber < 1 || levelNumber > CONFIG.TOTAL_LEVELS) return; gameState.currentLevel = levelNumber; localStorage.setItem('lastSelectedLevel', levelNumber); gameState.clearedPercentage = 0.0; gameState.startTime = Date.now(); gameState.isLevelComplete = false; setupFogCanvas(levelNumber); console.log(`Fase ${levelNumber} iniciada com sucesso.`); } ------------------------------------------------------------------------------- 4. LÓGICA DE LIMPEZA DO VIDRO & MECÂNICA DA FASE DE NEVE (2 ETAPAS) ------------------------------------------------------------------------------- // Estrutura de dados da malha de cobertura do vidro let gridMap = []; // Array de células com estados: 'fog', 'frost', 'cleared' function setupFogCanvas(levelNumber) { const isSnowLevel = CONFIG.SNOW_LEVELS.includes(levelNumber); const gridCols = 50; const gridRows = 100; gridMap = []; for (let r = 0; r < gridRows; r++) { let row = []; for (let c = 0; c < gridCols; c++) { row.push({ state: isSnowLevel ? 'fog' : 'fog', // 'fog' -> 'frost' -> 'cleared' opacity: 1.0 }); } gridMap.push(row); } } function handleTouchMove(touchX, touchY) { if (gameState.isLevelComplete) return; const isSnowLevel = CONFIG.SNOW_LEVELS.includes(gameState.currentLevel); const affectedCells = getCellsInRadius(touchX, touchY, CONFIG.BRUSH_RADIUS); affectedCells.forEach(cell => { if (isSnowLevel) { // Mecânica de Neve: Requer 2x passadas do dedo if (cell.state === 'fog') { cell.state = 'frost'; // 1ª passada: Apenas clareia cell.opacity = 0.4; } else if (cell.state === 'frost') { cell.state = 'cleared'; // 2ª passada: Limpa de vez cell.opacity = 0.0; } } else { // Mecânica Padrão de Chuva / Neblina: Limpa na 1ª passada cell.state = 'cleared'; cell.opacity = 0.0; } }); calculateProgress(); } function calculateProgress() { let totalCells = gridMap.length * gridMap[0].length; let clearedCells = 0; for (let r = 0; r < gridMap.length; r++) { for (let c = 0; c < gridMap[r].length; c++) { if (gridMap[r][c].state === 'cleared') { clearedCells++; } } } gameState.clearedPercentage = clearedCells / totalCells; checkWinCondition(); } ------------------------------------------------------------------------------- 5. CONDIÇÃO DE VITÓRIA E "LIMPEZA PERFEITA" ------------------------------------------------------------------------------- function checkWinCondition() { if (gameState.isLevelComplete) return; // Condição de Vitória: Ajustado para 99% (0.99) if (gameState.clearedPercentage >= CONFIG.WIN_THRESHOLD) { gameState.isLevelComplete = true; const timeTaken = (Date.now() - gameState.startTime) / 1000; // Verifica se cumpre o critério RARO de "Limpeza Perfeita" if (gameState.clearedPercentage >= CONFIG.PERFECT_CLEAR_THRESHOLD && timeTaken <= CONFIG.PERFECT_CLEAR_TIME_LIMIT) { triggerPerfectClearReward(); } onLevelComplete(); } } function triggerPerfectClearReward() { console.log("★ LIMPEZA PERFEITA! ★"); // Exibe efeito raro, áudio de recompensa especial e pontuação bônus showSpecialUIEffect("LIMPEZA PERFEITA!"); } ------------------------------------------------------------------------------- 6. FLUXO DE ANÚNCIOS (ADS) E TRANSIÇÕES DE TELA ------------------------------------------------------------------------------- function showAd() { return new Promise((resolve) => { console.log("Exibindo Anúncio Intersticial..."); // Integração com SDK de Ads (e.g. AdMob / Unity Ads) setTimeout(() => { console.log("Anúncio Concluído."); resolve(); }, 1500); // Simulação do término do anúncio }); } function onLevelComplete() { // Atualiza progresso salvo if (gameState.currentLevel === gameState.unlockedLevels && gameState.unlockedLevels < CONFIG.TOTAL_LEVELS) { gameState.unlockedLevels++; localStorage.setItem('unlockedLevels', gameState.unlockedLevels); } // Exibe Anúncio ao passar de fase showAd().then(() => { if (gameState.currentLevel < CONFIG.TOTAL_LEVELS) { let nextLvl = gameState.currentLevel + 1; localStorage.setItem('lastSelectedLevel', nextLvl); startLevel(nextLvl); } else { returnToMainMenu(); } }); } function returnToMainMenu() { // Exibe Anúncio ao voltar pro menu principal showAd().then(() => { initMainMenu(); showScreen('main-menu-screen'); }); } ------------------------------------------------------------------------------- 7.
